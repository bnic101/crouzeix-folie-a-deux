\begin{frame}{Spectral sets}
  \textbf{Def.} A set $S\subseteq\C$ is called a $K$-spectral set for an operator $T\in\B(\Hil)$ if
  \[
    \norm{p(T)}\le K\sup_{z\in S}|p(z)|\qquad\forall\,p\in\C[z].
  \]
  ($S$ bounded, polynomially convex.)

  \medskip
  \textbf{von Neumann's inequality.} The disc is a spectral set for any contraction:
  \[
    \norm{p(T)}\le\sup_{z\in\D}|p(z)|,\qquad \norm{T}\le1.
  \]
  (Also the closed half plane if $\re T=\tfrac{T+T^*}{2}\ge0$.)

  \medskip
  \begin{tabular}{@{}l|l@{}}
    von Neumann & Sz.-Nagy\\
    function theory & dilation theory\\
    Blaschke products & unitary dilation of $T$
  \end{tabular}
\end{frame}
\note{Two proofs of von Neumann. Same split for Crouzeix.}

\begin{frame}{Crouzeix conjecture}
  \begin{block}{Conjecture (Crouzeix 2004)}
    Let $T$ be a bounded operator on a Hilbert space $\Hil$. The numerical range $\overline{W(T)}$ is a $2$-spectral set for $T$.
  \end{block}

  \medskip
  \textbf{Remark.} This conjectured bound is sharp:
  \[
    W\!\begin{pmatrix}0&1\\0&0\end{pmatrix}=D(0,\tfrac12),\qquad p(z)=z,
  \]
  \[
    \Big\|\begin{pmatrix}0&1\\0&0\end{pmatrix}\Big\|=1=2\cdot\tfrac12=2\sup_{D(0,1/2)}|p(z)|.
  \]

  \medskip
  \textbf{Antecedents.}
  \begin{itemize}
    \item von Neumann, only for contractions.
    \item Okubo--Ando 1975, Berger--Stampfli 1967: the disc is a $2$-spectral set.
  \end{itemize}
\end{frame}
\note{Berger: $w(T)\le1$ iff $T^n=2V^*U^nV$. Crouzeix 2004: $2\times2$ case, equality only for disks.}

\begin{frame}{Double layer potential}
  Let $\Omega\subset\C$ be an open bounded convex set with smooth boundary $\partial\Omega$ and $\overline{W(T)}\subset\Omega$.
  $A(\Omega)$: holomorphic on $\Omega$, continuous on $\overline\Omega$, sup norm. $C(\partial\Omega)$: continuous on $\partial\Omega$, sup norm.
  For $\sigma\in\partial\Omega$ denote the outward pointing normal by $\nu=\nu(\sigma)$, and if $z\in\Omega$ define the double layer potential by
  \[
    \mu(\sigma,z)=\frac{1}{2\pi}\Big(\underbrace{\frac{\nu}{\sigma-z}}_{\text{first layer}}+\underbrace{\frac{\bar\nu}{\bar\sigma-\bar z}}_{\text{second layer}}\Big)
  \]
  $=\frac1\pi\re\frac{\nu}{\sigma-z}$. First layer: the Cauchy kernel.

\end{frame}

\begin{frame}{Double layer potential, cont.}
  Let
  \[
    C(f,T)=\frac{1}{2\pi i}\int_{\partial\Omega}f(\sigma)(\sigma-T)^{-1}\,d\sigma\in\B(\Hil)
  \]
  and
  \[
    S(f,T)=C(f,T)+C(\bar f,T)^*\in\B(\Hil).
  \]
  By definition $S$ is the function calculus integration against the double layer potential $\mu(\sigma,z)$, i.e.
  \[
    S(f,T)=\int_{\partial\Omega}f(\sigma)\,\mu(\sigma,T)\,ds .
  \]
  \begin{lemma}[Crouzeix--Palencia 2017]
    For $f\in C(\partial\Omega)$, $\ \norm{S(f,T)}=\norm{f(T)+C(\bar f,T)^*}\le2\norm{f}_{\partial\Omega}$.
  \end{lemma}
  So the question we must answer is: how does $\norm{S(f,T)}$ control $\norm{f(T)}$? In particular, can we estimate $C(\bar f,T)^*$ or make it vanish all together?
\end{frame}
\note{$\mu(\sigma,T)=\frac1\pi\re[\nu(\sigma-T)^{-1}]\ge0$: $\ip{Tx}{x}\in\Omega$, conjugate by resolvent. Mass $2I$: Cauchy. Lemma: Cauchy--Schwarz.}

\begin{frame}{Positivity and the double layer formula}
  Consider
  \[
    2\Phi(f):=f(T)+C(\bar f,T)^*=f(T)+\alpha(f)(T)^*,
  \]
  where
  \[
    \alpha(f)(z)=\frac{1}{2\pi i}\int_{\partial\Omega}\frac{\overline{f(\sigma)}}{\sigma-z}\,d\sigma .
  \]
  Note that $\alpha:A(\Omega)\to A(\Omega)$ is antilinear and $\alpha(1)=1$.

  \medskip
  Positivity of $\Phi$ says $2\norm{\Phi(f)}\le2\norm{f}_{\partial\Omega}$: the result from the previous slide, found in Crouzeix--Palencia.

  \medskip
  Scalar case: $2\Phi(f)(z)=f(z)+\overline{\alpha(f)(z)}$. Disc: $\alpha(f)=\overline{f(0)}$.

  \medskip
  An alternate approach (not taken by either ChatGPT proof) is to ``hack'' the formula by studying extremal functions: $f$ extremal, $\norm{f(T)}\le\norm{2\Phi(f)}$, which proves Crouzeix. (Bickel, Gorkin, Greenbaum, Overton.)
\end{frame}
\note{Website: $f(z)$ plus companion as vectors. Extremal $f$ is a Blaschke product (Crouzeix 2004).}

\begin{frame}{Crouzeix--Palencia, $1+\sqrt2$}
  $K=C_\Omega$, $\norm{f}\le1$, $T=f(A)$, $g=\alpha(f)$, $\norm{g}\le\norm{f}$ ($\Omega$ convex).
  \[
    TT^*TT^*=T\,\big(T+g(A)^*\big)^*\,TT^*-(fgf)(A)\,T^*
  \]
  \[
    K^4\le2K^3+K^2,\qquad K\le1+\sqrt2 .
  \]
  \hfill(Ransford--Schwenninger)

  \medskip
  Uses only $\norm{f(A)+g(A)^*}\le2$ and $\norm g\le\norm f$. Sharp under these (Ransford--Schwenninger 2018):
  \[
    T=\begin{pmatrix}1&1\\0&0\end{pmatrix},\qquad \Omega=D(0,\tfrac14)\cup D(1,\tfrac14),
  \]
  $g=-f(0)$, $-f(1)$ on the two discs; $h=\mp1$: $\norm{h(T)}=1+\sqrt2$.

  \medskip
  Here $g(1)=-1$. The Cauchy transform: $\alpha(1)=1$.
\end{frame}
\note{Insert $T^*=(T+g(A)^*)^*-g(A)$ in the middle. $f(T)=\begin{pmatrix}f(1)&f(1)-f(0)\\0&f(0)\end{pmatrix}$; $h(T)=\begin{pmatrix}1&2\\0&-1\end{pmatrix}$. Not a counterexample: $W(T)\not\subset\Omega$.}

\begin{frame}{Equivalent formulation}
  \begin{theorem}[Twisted real part]
    Let $\A$ be a unital commutative operator algebra, let $\theta:\A\to\B(\Hil)$ be a bounded homomorphism and let $\alpha:\A\to\A$ be a unital antilinear map such that the unital linear map
    \[
      \theta_\alpha:\A\to\B(\Hil),\qquad a\mapsto\tfrac12\big(\theta(a)+\theta(\alpha(a))^*\big)
    \]
    is contractive. Then $\norm{\theta}\le2$.
  \end{theorem}
  \begin{itemize}
    \item Ransford--Schwenninger (2018) show that the twisted real part theorem $\Rightarrow$ Crouzeix.
    \item Michael Hartz (2026) proves the theorem using Jin's ChatGPT solution.
  \end{itemize}
\end{frame}
\note{$\theta(f)=f(T)$, $\alpha$ = Cauchy transform, $\theta_\alpha=\Phi$. Unital contractive $\Rightarrow$ positive.}

\begin{frame}{Function theory}
  Jin's GitHub; unpublished notes by Hartz.
  \begin{theorem}[Holomorphic $\to$ resolvent transfer]
    Let $T\in\B(\Hil)$ with $\sigma(T)\subset\overline\D$. Let $H:\D\to\B(\Hil)$ be a holomorphic function such that $H(0)=I$ and $\re H(z)\ge0$ for all $z\in\D$. Assume that
    \[
      H(z)-(I-zT)^{-1}
    \]
    commutes with $T^*$ for all $z\in\D$. Then $\norm{T}\le2$.
  \end{theorem}
  \begin{itemize}
    \item Hartz shows that holomorphic $\to$ resolvent transfer $\Rightarrow$ twisted real part.
    \item $H(z)-(I-zT)^{-1}=0$: Okubo--Ando.
  \end{itemize}
\end{frame}
\note{$\re(I-zT)^{-1}\ge0$ iff $w(T)\le1$ (class $C_2$). Transfer $\Rightarrow$ TRP: $c_z=(1+za)(1-za)^{-1}$, $H(z)=\theta_\alpha(c_z)$, $\theta(c_z)=2(I-zT)^{-1}-I$, defect $=\frac12(\theta(\alpha(c_z))^*-I)$. Concretely $\Delta(w)=\sum_{m\ge1}w^m\alpha(f^m)(T)^*$.}

%======================================================================
\begin{frame}{Proof}
  $\Delta(z):=H(z)-(I-zT)^{-1}$, $R(z):=(I-zT)^{-1}$. On $H^2\otimes\Hil$: $\re M_H\ge0$, $M_H=M_R+M_\Delta$.

  \medskip
  $N_W:=\{F=\sum F_nz^n:\ \sum W^nF_n=0\}$, $\sigma(W)\subset\D$.
  \begin{itemize}
    \item $\Delta(z)W=W\Delta(z)$ $\Rightarrow$ $M_\Delta N_W\subset N_W$.
    \item $\Delta(0)=0$, $\Gamma$ the kernel of $N_W$ at $0$ $\Rightarrow$ $\Gamma^*M_\Delta\Gamma=0$.
  \end{itemize}
  $W=\tfrac12T^*$:
  \[
    \Gamma^*(M_R+M_R^*)\Gamma=\Gamma^*(M_H+M_H^*)\Gamma\ge0 .
  \]
  $P=\sum4^{-n}T^{*n}T^n$, $Q=\sum2^{-n}T^{*n}T^n$, $Y=Q-P\ge\tfrac14T^*T$:
  \[
    4Y-YP-PY\ge0\ \Rightarrow\ P\le2I\ \Rightarrow\ T^*T\le T^*PT=4(P-I)\le4I .
  \]
  Jin: sample the Herglotz kernel of $H$ at $\lambda_i/2$ and at $0$.
\end{frame}
\note{Compress to functions vanishing at the operator point $W$. Defect removed exactly. Appendix: computation of $\Gamma$, eigenvector argument, Stein identity $P=I+\frac14T^*PT$.}

\begin{frame}{Dilation theory}
  \begin{lemma}[Lorist--Schwenninger, ChatGPT]
    Let $\Hil$ be a Hilbert space and $T\in\B(\Hil)$. Suppose there exists a Hilbert space $\K$, a contraction $Q\in\B(\K)$ and an isometry $V\in\B(\Hil,\K)$ such that the operators
    \[
      E_n:=2V^*Q^{*n}V-T^{*n},\qquad n\in\mathbb N,
    \]
    are uniformly bounded and commute with $T$. Then $\norm{T}\le2$.
  \end{lemma}
  \begin{itemize}
    \item $E_n=0$: $2$-dilation of $T^*$, $\norm{T}\le2$.
    \item $\K=L^2(\partial\Omega,ds;\Hil)$, $Vx=\tfrac1{\sqrt2}\mu(\cdot,T)^{1/2}x$, $Q=M_f$, $\norm f_\infty=1$, $T=f(A)$:
      \[
        V^*Q^nV=\Phi(f^n),\qquad E_n=2\Phi(f^n)^*-f(A)^{*n}=\alpha(f^n)(A).
      \]
    \item $E_n$ commutes with $f(A)$; $\norm{E_n}\le\norm\theta\norm\alpha$.
  \end{itemize}
\end{frame}
\note{Stinespring for $\Phi$. $V$ isometry since $\int\mu=2I$. $\norm\alpha\le1$ not used. Abstract: $\theta_\alpha$ completely positive $\Rightarrow\norm\theta\le2$.}

%======================================================================
\begin{frame}{Proof}
  $\kappa:=\norm{T}>1$, $T^*Tx=\kappa^2x$, $\norm x=1$ (ultrapower).
  \[
    m_n:=\re\ip{E_nT^nx}{x},\qquad u:=Q^*VTx-\kappa Vx .
  \]
  \begin{align*}
    \norm{u}^2&\le 2\kappa^2-\kappa^3-\kappa m_1 && (n=1)\\[2pt]
    \kappa m_1&\ge-\frac{\norm{u}^2}{(\kappa-1)^2} && (\text{telescope } \kappa m_n-m_{n+1},\ \text{all } n)
  \end{align*}
  \[
    \norm{u}^2\big(1-(\kappa-1)^{-2}\big)\le\kappa^2(2-\kappa) .
  \]

  \medskip
  First line alone: $\kappa\le1+\sqrt{1-m_1}$. $\norm\alpha\le1\Rightarrow m_1\ge-1\Rightarrow1+\sqrt2$. $m_1<0$ for some $3\times3$.
\end{frame}
\note{$u$: failure of $VTx$ to be a $\kappa$-eigenvector of $Q^*$. $m_1=\re\ip{(f\alpha(f))(A)x}{x}$. Appendix: both estimates.}

\begin{frame}{Appendix: $\Gamma$}
  $K_W:=E_W^*$, $(K_Wx)(z)=(I-zW^*)^{-1}x$, $K_0x\equiv x$. $W=\tfrac12T^*$, $P=K_W^*K_W\ge I$, $K_WP^{-1}K_W^*$ = projection onto $\ran K_W=N_W^\perp$.
  \[
    \Gamma=(I-K_WP^{-1}K_W^*)K_0=K_0-K_WP^{-1}\qquad(K_W^*K_0=I).
  \]
  $(I-zT)^{-1}(I-zT/2)^{-1}=2(I-zT)^{-1}-(I-zT/2)^{-1}$:
  \[
    M_RK_W=2K_{T^*}-K_W,\qquad M_RK_0=K_{T^*},\qquad M_R^*K_0=K_0 .
  \]
  $Q=K_W^*K_{T^*}$:
  \[
    \begin{pmatrix}K_W^*\\K_0^*\end{pmatrix}(M_R+M_R^*)\begin{pmatrix}K_W&K_0\end{pmatrix}
    =\begin{pmatrix}4Q-2P&Q+I\\Q+I&2I\end{pmatrix},
  \]
  \[
    \Gamma^*(M_R+M_R^*)\Gamma=P^{-1}(4Y-YP-PY)P^{-1}.
  \]
\end{frame}

\begin{frame}{Appendix: $4Y-YP-PY\ge0\Rightarrow\norm{T}\le2$}
  $P=\sum_{n\ge0}4^{-n}T^{*n}T^n\ge I$, $Y\ge\tfrac14T^*T\ge0$.

  \medskip
  $Px=\alpha x$, $\norm x=1$, $\alpha>2$:
  \[
    0\le\ip{(4Y-YP-PY)x}{x}=(4-2\alpha)\ip{Yx}{x},
  \]
  so $Yx=0$, so $Tx=0$, so $Px=x$.
  (Approximate eigenvectors, or ultrapower.) So $I\le P\le2I$.

  \medskip
  $P=I+\tfrac14T^*PT$:
  \[
    T^*T\le T^*PT=4(P-I)\le4I .
  \]
\end{frame}

\begin{frame}{Appendix: the two estimates}
  $E_nT=TE_n$, $T^*Tx=\kappa^2x$:
  \begin{align*}
    \kappa m_n-m_{n+1}
    &=\re\ip{T^n(\kappa E_n-E_{n+1}T)x}{x}\\
    &=\kappa(\kappa-1)\norm{T^{*n}x}^2-2\re\ip{V^*Q^{*n}u}{T^{*n}x}\\
    &=\kappa(\kappa-1)\Big\|T^{*n}x-\frac{V^*Q^{*n}u}{\kappa(\kappa-1)}\Big\|^2-\frac{\norm{V^*Q^{*n}u}^2}{\kappa(\kappa-1)}\\
    &\ge-\frac{\norm{u}^2}{\kappa(\kappa-1)} .
  \end{align*}
  Multiply by $\kappa^{-n+1}$, sum over $n\ge1$; $(m_n)$ bounded, so the sum telescopes to $\kappa m_1$.

  \medskip
  $\norm{u}^2=\norm{Q^*VTx}^2+\kappa^2-2\kappa\re\ip{V^*Q^*VTx}{x}$, $\norm{Q^*VTx}\le\kappa$, $2V^*Q^*V=T^*+E_1$:
  \[
    \norm{u}^2\le2\kappa^2-\kappa\re\ip{T^*Tx}{x}-\kappa m_1=2\kappa^2-\kappa^3-\kappa m_1 .
  \]
\end{frame}
