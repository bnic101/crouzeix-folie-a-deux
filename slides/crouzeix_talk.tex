\begin{frame}{Numerical range}
  \textbf{Def.} Let $\Hil$ be a complex Hilbert space and $T\in\B(\Hil)$ a bounded operator. The numerical range of $T$ is
  \[
    W(T)=\{\ip{Tx}{x}:\ x\in\Hil,\ \norm{x}=1\}.
  \]
  In finite dimension closed and convex (Toeplitz--Hausdorff). In infinite dimension convex, $\overline{W(T)}\supseteq\conv\sigma(T)$.

\begin{frame}{Spectral sets}
  \textbf{Def.} A set $S\subseteq\C$ is called a $K$-spectral set for an operator $T\in\B(\Hil)$ if
  \[
    \norm{p(T)}\le K\sup_{z\in S}|p(z)|\qquad\forall\,p\in\C[z].
  \]
  ($S$ bounded, polynomially convex.)

  \medskip
  \textbf{von Neumann's inequality.} The disc is a spectral set for any contraction:
  \[
    \norm{p(T)}\le\sup_{z\in\D}|p(z)|,\qquad \norm{T}\le1.
  \]
  (Also the closed half plane if $\re T=\tfrac{T+T^*}{2}\ge0$.)
\end{frame}
\note{[2 min] Spectral set: von Neumann's word.}

\begin{frame}{Crouzeix conjecture}
  \begin{block}{Conjecture (Crouzeix 2004)}
    Let $T$ be a bounded operator on a Hilbert space $\Hil$. The numerical range $\overline{W(T)}$ is a $2$-spectral set for $T$.
  \end{block}

  \medskip
  \textbf{Remark.} This conjectured bound is sharp:
  \[
    W\!\begin{pmatrix}0&1\\0&0\end{pmatrix}=D(0,\tfrac12),\qquad p(z)=z,
  \]
  \[
    \Big\|\begin{pmatrix}0&1\\0&0\end{pmatrix}\Big\|=1=2\cdot\tfrac12=2\sup_{D(0,1/2)}|p(z)|.
  \]
\end{frame}
\note{[2 min] $\ip{Nx}{x}=x_2\bar x_1$.}

%======================================================================
\begin{frame}{Antecedents}
  \large
  \begin{itemize}
    \item von Neumann: constant $1$, only for contractions.
    \item Okubo--Ando 1975, Berger--Stampfli 1967: the disc is a $2$-spectral set.
    \item Via $2$-dilations: $W(T)\subset\overline\D$ iff $T^n=2V^*U^nV$ (Berger).
    \item Crouzeix 2004: true for $2\times2$, equality only when $W(T)$ is a disc.
  \end{itemize}
\end{frame}
\note{[3 min] Constant $2$ for the disc is a theorem, not a triviality: Berger--Stampfli need $p(0)=0$, Okubo--Ando remove it. $2\times2$: exact constant as a function of the eccentricity.}

\begin{frame}{Constants}
  $\Omega$ bounded convex, smooth boundary, $W(T)\subset\Omega$.
  \[
    C_\Omega:=\sup\big\{\norm{f(T)}:\ W(T)\subset\Omega,\ \norm{f}_\infty\le1\big\}.
  \]
  {\small\setlength{\tabcolsep}{3pt}
  \begin{tabular}{@{}llll@{}}
    Delyon--Delyon & 1999 & $C_\Omega<\infty$ & depends on $\Omega$\\
    Crouzeix & 2004 & $2$ & only for $2\times2$\\
    Crouzeix & 2007 & $11.08$ & uniform in $\Omega$\\
    Crouzeix--Palencia & 2017 & $1+\sqrt2$ & completely bounded\\
    Jin & 27 July 2026 & $2$ & function theory\\
    Lorist--Schwenninger & Aug 2026 & $2$ & dilation theory
  \end{tabular}}

  \medskip
  \textbf{Technique} (all of the above): the double layer potential.
\end{frame}
\note{[2 min] Delyon--Delyon: constant from diameter$^2$/area. Crouzeix 2007: hard estimates on the Cauchy transform. Crouzeix--Palencia: six pages, survives matrix-valued $f$. Nine years, then two proofs in a week.}

\begin{frame}{Double layer potential}
  $\Omega\subset\C$ open bounded convex, smooth boundary, $\overline{W(T)}\subset\Omega$. $A(\Omega)$: holomorphic on $\Omega$, continuous on $\overline\Omega$. $C(\partial\Omega)$: continuous on $\partial\Omega$. Sup norms.
  For $\sigma\in\partial\Omega$ let $\nu=\nu(\sigma)$ be the outward pointing normal; for $z\in\Omega$ the double layer potential is
  \[
    \mu(\sigma,z)=\frac{1}{2\pi}\Big(\underbrace{\frac{\nu}{\sigma-z}}_{\text{first layer}}+\underbrace{\frac{\bar\nu}{\bar\sigma-\bar z}}_{\text{second layer}}\Big)
  \]
  $=\frac1\pi\re\frac{\nu}{\sigma-z}$. First layer: the Cauchy kernel.
\end{frame}
\note{[2 min] Twice the real part of the Cauchy kernel.}

\begin{frame}{Double layer potential, cont.}
  Let
  \[
    C(f,T)=\frac{1}{2\pi i}\int_{\partial\Omega}f(\sigma)(\sigma-T)^{-1}\,d\sigma\in\B(\Hil)
  \]
  and
  \[
    S(f,T)=C(f,T)+C(\bar f,T)^*\in\B(\Hil).
  \]
  By definition $S$ is the function calculus integration against the double layer potential $\mu(\sigma,z)$, i.e.
  \[
    S(f,T)=\int_{\partial\Omega}f(\sigma)\,\mu(\sigma,T)\,ds,
  \]
  with $\mu(\sigma,T)=\tfrac1\pi\re[\nu(\sigma)(\sigma-T)^{-1}]\ge0$, $\int\mu\,ds=2I$.
\end{frame}
\note{[2 min] $\mu(\sigma,T)\ge0$: $\ip{Tx}{x}\in\Omega$ gives $\re\bar\nu(\sigma-T)\ge0$; conjugate by the resolvent. Mass $2I$: Cauchy's formula plus its adjoint.}

%======================================================================
\begin{frame}{The question}
  \begin{lemma}[Crouzeix--Palencia 2017]
    For $f\in C(\partial\Omega)$,
    \[
      \norm{S(f,T)}=\norm{f(T)+C(\bar f,T)^*}\le2\norm{f}_{\partial\Omega}.
    \]
  \end{lemma}
  Proof: $\mu\ge0$, $\int\mu\,ds=2I$, Cauchy--Schwarz.

  \bigskip
  So the question we must answer is: how does $\norm{S(f,T)}$ control $\norm{f(T)}$? In particular, can we estimate $C(\bar f,T)^*$ or make it vanish all together?
\end{frame}
\note{[3 min] $\mu(\sigma,T)\ge0$: $\ip{Tx}{x}\in\Omega$ gives $\re\bar\nu(\sigma-T)\ge0$; conjugate by the resolvent. Mass $2I$: Cauchy's formula plus its adjoint. First layer of $S(f,T)$ is $f(T)$; second layer is the adjoint of the Cauchy integral of $\bar f$, not of $f$.}

%======================================================================
\begin{frame}{Positivity and the double layer formula}
  Consider
  \[
    2\Phi(f):=f(T)+C(\bar f,T)^*=f(T)+\alpha(f)(T)^*,
  \]
  where
  \[
    \alpha(f)(z)=\frac{1}{2\pi i}\int_{\partial\Omega}\frac{\overline{f(\sigma)}}{\sigma-z}\,d\sigma .
  \]
  Note that $\alpha:A(\Omega)\to A(\Omega)$ is antilinear and $\alpha(1)=1$.

  \medskip
  Positivity of $\Phi$ says $2\norm{\Phi(f)}\le2\norm{f}_{\partial\Omega}$: the result from the previous slide, found in Crouzeix--Palencia.
\end{frame}
\note{[2 min] Same identity, $\Phi=\frac12S$.}

%======================================================================
\begin{frame}{Positivity and the double layer formula}
  Scalar case: $2\Phi(f)(z)=f(z)+\overline{\alpha(f)(z)}$, holomorphic plus anti-holomorphic.

  \medskip
  Disc: $\alpha(f)=\overline{f(0)}$, a constant.

  \medskip
  Crouzeix--Palencia: estimate the companion, $\norm{\alpha(f)}\le\norm{f}$. Gives $1+\sqrt2$.

  \medskip
  An alternate approach (not taken by either ChatGPT proof) is to ``hack'' the formula by studying extremal functions: $f$ extremal, $\norm{f(T)}\le\norm{2\Phi(f)}$, which proves Crouzeix. (Bickel, Gorkin, Greenbaum, Overton.)
\end{frame}
\note{[3 min] Website: $f(z)$ and the companion as vectors; $\Phi(f)(z)$ stays in the convex hull of $f(\partial\Omega)$. Disc: companion constant, so Berger--Stampfli is the identity read off. Extremal $f$ is a Blaschke product (Crouzeix 2004, Nevanlinna--Pick).}

\begin{frame}{Equivalent formulation}
  \begin{theorem}[Twisted real part]
    Let $\A$ be a unital commutative operator algebra, let $\theta:\A\to\B(\Hil)$ be a bounded homomorphism and let $\alpha:\A\to\A$ be a unital antilinear map such that the unital linear map
    \[
      \theta_\alpha:\A\to\B(\Hil),\qquad a\mapsto\tfrac12\big(\theta(a)+\theta(\alpha(a))^*\big)
    \]
    is contractive. Then $\norm{\theta}\le2$.
  \end{theorem}
  \begin{itemize}
    \item Ransford--Schwenninger (2018): twisted real part $\Rightarrow$ Crouzeix.
    \item Michael Hartz (2026) proves the theorem using Jin's ChatGPT solution.
  \end{itemize}
\end{frame}
\note{[2 min] $\theta(f)=f(T)$, $\alpha$ the Cauchy transform: $\theta_\alpha=\Phi$. Unital and contractive $\Rightarrow$ positive. Both proofs establish this: Hartz as stated, Lorist--Schwenninger with ``completely positive''.}

\begin{frame}{Function theory}
  \begin{theorem}[Holomorphic $\to$ resolvent transfer; Jin, operator form Hartz]
    Let $T\in\B(\Hil)$ with $\sigma(T)\subset\overline\D$. Let $H:\D\to\B(\Hil)$ be a holomorphic function such that $H(0)=I$ and $\re H(z)\ge0$ for all $z\in\D$. Assume that
    \[
      H(z)-(I-zT)^{-1}
    \]
    commutes with $T^*$ for all $z\in\D$. Then $\norm{T}\le2$.
  \end{theorem}
  \medskip
  $H(z)-(I-zT)^{-1}=0$: $\re(I-zT)^{-1}\ge0$ iff $W(T)\subset\overline\D$ (Berger); conclusion is Okubo--Ando.
\end{frame}
\note{[3 min] $H$: Carath\'eodory function normalised at $0$. Class $C_\rho$: $(\rho-2)I+2\re(I-zT)^{-1}\ge0$; $\rho=2$ is $w(T)\le1$.}

%======================================================================
\begin{frame}{Transfer $\Rightarrow$ twisted real part (Hartz)}
  Let $a\in\A$, $\norm a\le1$, $T=\theta(a)$. For $z\in\D$ the Cayley transform $c_z=(1+za)(1-za)^{-1}$ has $\re c_z\ge0$, and $\theta_\alpha$ is positive, so
  \[
    H(z):=\theta_\alpha(c_z)
  \]
  is holomorphic with $H(0)=I$ and $\re H(z)\ge0$. Since $\theta$ is a homomorphism, $\theta(c_z)=2(I-zT)^{-1}-I$, hence
  \[
    H(z)-(I-zT)^{-1}=\tfrac12\big(\theta(\alpha(c_z))^*-I\big),
  \]
  the adjoint of an element of $\theta(\A)$, which is commutative and contains $T$. So $H(z)-(I-zT)^{-1}$ commutes with $T^*$, and the transfer theorem gives $\norm{\theta(a)}\le2$.
\end{frame}
\note{[4 min] $\re c_z=(1-za)^{-*}(I-|z|^2a^*a)(1-za)^{-1}\ge0$. For $\theta(f)=f(T)$, expanding $c_w$ in powers of $f$: $H(w)-(I-wT)^{-1}=\sum_{m\ge1}w^m\alpha(f^m)(T)^*$; the identity for all powers of $f$. Crouzeix--Palencia used it once.}

%======================================================================
\begin{frame}{Proof of the transfer theorem, I}
  $R(z)=(I-zT)^{-1}$, $\Delta=H-R$. On $H^2\otimes\Hil$ ($\Hil$-valued power series), $\re H\ge0$ gives $\re M_H\ge0$, and $M_H=M_R+M_\Delta$.

  \medskip
  For $W$ with $\sigma(W)\subset\D$, the functions vanishing at $W$:
  \[
    N_W=\{F=\textstyle\sum F_nz^n:\ \sum W^nF_n=0\},
  \]
  and $\Gamma:\Hil\to N_W$ its reproducing kernel at $0$.
  \begin{itemize}
    \item If $\Delta(z)$ commutes with $W$, then $M_\Delta N_W\subset N_W$.
    \item Since $\Delta(0)=0$, $\Gamma^*M_\Delta\Gamma=0$.
  \end{itemize}
  Take $W=\tfrac12T^*$. Then
  \[
    \Gamma^*(M_R+M_R^*)\Gamma=\Gamma^*(M_H+M_H^*)\Gamma\ge0 .
  \]
\end{frame}
\note{[3 min] Functions vanishing at an operator point form an ``ideal'' when $\Delta$ commutes with $W$; kernel at $0$ kills $M_\Delta$ since $\Delta(0)=0$. Defect removed exactly, not estimated.}

%======================================================================
\begin{frame}{Proof of the transfer theorem, II}
  Explicitly $\Gamma=K_0-K_WP^{-1}$, with $K_W$ the kernel map at $W$,
  \[
    P=K_W^*K_W=\sum_{n\ge0}4^{-n}T^{*n}T^n,
  \]
  \[
    Q=K_W^*K_{T^*}=\sum_{n\ge0}2^{-n}T^{*n}T^n .
  \]
  The resolvent identity turns the last inequality into
  \[
    4Y-YP-PY\ge0,\qquad Y=Q-P\ge\tfrac14T^*T .
  \]
  If $Px=\alpha x$ with $\alpha>2$, testing on $x$ gives $(4-2\alpha)\ip{Yx}{x}\ge0$, so $Yx=0$, so $Tx=0$, so $Px=x$. Hence $P\le2I$. With $P=I+\tfrac14T^*PT$,
  \[
    T^*T\le T^*PT=4(P-I)\le4I .
  \]
\end{frame}
\note{[4 min] Functions vanishing at an operator point form an ``ideal'' when $\Delta$ commutes with $W$; kernel at $0$ kills $M_\Delta$ since $\Delta(0)=0$. Defect removed exactly, not estimated. Weights $4^{-n},2^{-n}$ from sampling radius $\frac12$. Jin does this in coordinates: sample the Herglotz kernel at $\lambda_i/2$ and at $0$. Appendix: $\Gamma=K_0-K_WP^{-1}$, eigenvector argument, Stein $P=I+\frac14T^*PT$.}

\begin{frame}{Dilation theory}
  \begin{lemma}[Lorist--Schwenninger, ChatGPT]
    Let $\Hil$ be a Hilbert space and $T\in\B(\Hil)$. Suppose there exists a Hilbert space $\K$, a contraction $Q\in\B(\K)$ and an isometry $V\in\B(\Hil,\K)$ such that the operators
    \[
      E_n:=2V^*Q^{*n}V-T^{*n},\qquad n\in\mathbb N,
    \]
    are uniformly bounded and commute with $T$. Then $\norm{T}\le2$.
  \end{lemma}
  \medskip
  $E_n=0$: $2$-dilation of $T^*$, $\norm{T}\le2$.
\end{frame}
\note{[2 min] Perturbed $2$-dilation; error of any size, bounded in $n$, commuting with $T$.}

%======================================================================
\begin{frame}{Lemma $\Rightarrow$ Crouzeix}
  Let $\K=L^2(\partial\Omega,ds;\Hil)$ and $Vx=\tfrac1{\sqrt2}\mu(\cdot,A)^{1/2}x$; $V$ is an isometry because $\int\mu\,ds=2I$. For $\norm f_\infty=1$ let $Q$ be multiplication by $f$, a contraction. Then
  \[
    V^*Q^nV=\tfrac12\int f^n\mu\,ds=\Phi(f^n),
  \]
  and the identity for $f^n$, adjointed, gives with $T=f(A)$
  \[
    E_n=2\Phi(f^n)^*-f(A)^{*n}=\alpha(f^n)(A).
  \]
  $E_n$ commutes with $f(A)$ (both come from the functional calculus) and $\norm{E_n}\le\norm\theta\norm\alpha$. The lemma gives $\norm{f(A)}\le2$. The bound $\norm\alpha\le1$ is not used.
\end{frame}
\note{[4 min] A genuine dilation of $\Phi$, a perturbed $2$-dilation of $f(A)$, error exactly the companion. Commutes: same functional calculus. Bounded: Crouzeix 2007 had $9.08$. Abstract form: $\theta_\alpha$ completely positive $\Rightarrow\norm\theta\le2$ (Arveson, Stinespring).}

%======================================================================
\begin{frame}{Proof of the lemma, I}
  Suppose $\kappa=\norm T>1$ and take a unit $x$ with $T^*Tx=\kappa^2x$ (ultrapower). Put $m_n=\re\ip{E_nT^nx}{x}$ and
  \[
    u=Q^*VTx-\kappa Vx,
  \]
  the failure of $VTx$ to be a $\kappa$-eigenvector of $Q^*$.

  \medskip
  From the identity for $n=1$, $V$ isometric and $Q$ contractive:
  \[
    \norm{u}^2\le 2\kappa^2-\kappa^3-\kappa m_1 .
  \]
  This alone gives $\kappa\le1+\sqrt{1-m_1}$; $m_1\ge-1$ gives $1+\sqrt2$, and $m_1<0$ for some $3\times3$ matrices.
\end{frame}
\note{[2 min] $m_1=\re\ip{(f\alpha(f))(A)x}{x}$, central since Crouzeix--Palencia. $m_1\ge0$ would finish; false in general.}

%======================================================================
\begin{frame}{Proof of the lemma, II}
  From the identities for all $n$ and $E_nT=TE_n$, completing a square,
  \[
    \kappa m_n-m_{n+1}\ge-\frac{\norm u^2}{\kappa(\kappa-1)}\qquad(n\ge1).
  \]
  Multiply by $\kappa^{-n+1}$ and sum; the left side telescopes:
  \[
    \kappa m_1\ge-\frac{\norm{u}^2}{(\kappa-1)^2} .
  \]
  With the upper bound,
  \[
    \norm{u}^2\big(1-(\kappa-1)^{-2}\big)\le\kappa^2(2-\kappa),
  \]
  impossible for $\kappa>2$.
\end{frame}
\note{[4 min] $u$: failure of $VTx$ to be a $\kappa$-eigenvector of $Q^*$. Upper bound: $n=1$ only; $m_1=\re\ip{(f\alpha(f))(A)x}{x}$, central since Crouzeix--Palencia. Lower bound: commutation plus the identity for $n+1$, summed with weights $\kappa^{-n+1}$. $m_1\ge0$ would finish; false in general. Appendix: both estimates.}

%======================================================================
\begin{frame}{Comparison}
  \small
  \begin{tabular}{@{}lll@{}}
     & function theory & dilation theory\\
    \hline\\[-8pt]
    positivity & $\re H\ge0$ & $\Phi=V^*\pi(\cdot)V$\\[2pt]
    all powers & $H(w)=I+2\sum_m w^m\Phi(f^m)$ & $E_n$, all $n$\\[2pt]

\begin{frame}{Appendix: $4Y-YP-PY\ge0\Rightarrow\norm{T}\le2$}
  $P=\sum_{n\ge0}4^{-n}T^{*n}T^n\ge I$, $Y\ge\tfrac14T^*T\ge0$.

  \medskip
  $Px=\alpha x$, $\norm x=1$, $\alpha>2$:
  \[
    0\le\ip{(4Y-YP-PY)x}{x}=(4-2\alpha)\ip{Yx}{x},
  \]
  so $Yx=0$, so $Tx=0$, so $Px=x$.
  (Approximate eigenvectors, or ultrapower.) So $I\le P\le2I$.

  \medskip
  $P=I+\tfrac14T^*PT$:
  \[
    T^*T\le T^*PT=4(P-I)\le4I .
  \]
\end{frame}

\begin{frame}{Appendix: the two estimates}
  \small
  $E_nT=TE_n$, $T^*Tx=\kappa^2x$:
  \begin{align*}
    \kappa m_n-m_{n+1}
    &=\re\ip{T^n(\kappa E_n-E_{n+1}T)x}{x}\\
    &=\kappa(\kappa-1)\norm{T^{*n}x}^2-2\re\ip{V^*Q^{*n}u}{T^{*n}x}\\
    &=\kappa(\kappa-1)\Big\|T^{*n}x-\frac{V^*Q^{*n}u}{\kappa(\kappa-1)}\Big\|^2-\frac{\norm{V^*Q^{*n}u}^2}{\kappa(\kappa-1)}\\
    &\ge-\frac{\norm{u}^2}{\kappa(\kappa-1)} .
  \end{align*}
  Multiply by $\kappa^{-n+1}$, sum over $n\ge1$; $(m_n)$ bounded, so the sum telescopes to $\kappa m_1$.

  \medskip
  $\norm{u}^2=\norm{Q^*VTx}^2+\kappa^2-2\kappa\re\ip{V^*Q^*VTx}{x}$, $\norm{Q^*VTx}\le\kappa$, $2V^*Q^*V=T^*+E_1$:
  \[
    \norm{u}^2\le2\kappa^2-\kappa\re\ip{T^*Tx}{x}-\kappa m_1=2\kappa^2-\kappa^3-\kappa m_1 .
  \]
\end{frame}
